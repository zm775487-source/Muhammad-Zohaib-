Student Class + Add Record (Java)

// Student.java — model class
public class Student {
    private String roll, name, program, phone;
    public Student(String roll, String name,
                   String program, String phone) {
        this.roll = roll;  this.name = name;
        this.program = program; this.phone = phone;
    }
    // getters...
    public String getRoll() { return roll; }
    public String getName() { return name; }
}

Database (SQL)

-- students table
CREATE TABLE students (
    roll_no   VARCHAR(15) PRIMARY KEY,
    full_name VARCHAR(60) NOT NULL,
    program   VARCHAR(40),
    phone     VARCHAR(15)
);

-- search a student
SELECT * FROM students
WHERE full_name LIKE '%Ali%';

-- update a record
UPDATE students SET phone = '0300-1112223'
WHERE roll_no = 'BSCS-101';

-- delete a record
DELETE FROM students
WHERE roll_no = 'BSCS-101';
