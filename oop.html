// ============================================================
//  Library Management System - OOP Project in C++
//  Concepts: Encapsulation, Inheritance, Polymorphism,
//            Abstraction, Constructors, Virtual Functions
// ============================================================
#include <iostream>
#include <vector>
#include <string>
using namespace std;

// ---------- BOOK: base class (Encapsulation) ----------
class Book {
private:
    int id;
    string title, author;
    bool issued;
public:
    Book(int i, string t, string a)
        : id(i), title(t), author(a), issued(false) {}

    int    getId()     const { return id; }
    string getTitle()  const { return title; }
    bool   isIssued()  const { return issued; }
    void   issue()           { issued = true; }
    void   giveBack()        { issued = false; }

    // Virtual function -> run-time Polymorphism
    virtual void display() const {
        cout << id << " | " << title << " by " << author
             << (issued ? "  [ISSUED]" : "  [Available]") << "\n";
    }
    virtual ~Book() {}
};

// ---------- EBOOK: derived class (Inheritance) ----------
class EBook : public Book {
    double sizeMB;
public:
    EBook(int i, string t, string a, double s)
        : Book(i, t, a), sizeMB(s) {}

    void display() const override {          // Polymorphism
        Book::display();
        cout << "     (E-Book, " << sizeMB << " MB)\n";
    }
};

// ---------- PERSON: abstract class (Abstraction) ----------
class Person {
protected:
    string name;
public:
    Person(string n) : name(n) {}
    virtual void role() const = 0;           // pure virtual
    virtual ~Person() {}
};

class Member : public Person {
    int memberId;
public:
    Member(string n, int id) : Person(n), memberId(id) {}
    void role() const override {
        cout << name << " (Member #" << memberId << ")\n";
    }
};

// ---------- LIBRARY: manages everything ----------
class Library {
    vector<Book*> books;   // base pointers -> polymorphism
public:
    void addBook(Book* b) { books.push_back(b); }

    Book* find(int id) {
        for (Book* b : books)
            if (b->getId() == id) return b;
        return nullptr;
    }
    void issueBook(int id) {
        Book* b = find(id);
        if (b && !b->isIssued()) { b->issue();    cout << "Book issued.\n"; }
        else                        cout << "Book not available.\n";
    }
    void returnBook(int id) {
        Book* b = find(id);
        if (b) { b->giveBack(); cout << "Book returned.\n"; }
    }
    void showAll() const {
        for (Book* b : books) b->display();
    }
    ~Library() { for (Book* b : books) delete b; }
};

// ---------- MAIN: menu-driven program ----------
int main() {
    Library lib;
    lib.addBook(new Book (1, "C++ Primer",        "Lippman"));
    lib.addBook(new EBook(2, "Clean Code (PDF)",  "Martin", 4.2));
    lib.addBook(new Book (3, "Data Structures",   "Karumanchi"));

    Member m("Muhammad Zohaib", 101);
    int choice, id;
    do {
        cout << "\n1. Show Books  2. Issue  3. Return  0. Exit\nChoice: ";
        cin >> choice;
        if (choice == 1) lib.showAll();
        else if (choice == 2) { cout << "Book ID: "; cin >> id; lib.issueBook(id); }
        else if (choice == 3) { cout << "Book ID: "; cin >> id; lib.returnBook(id); }
    } while (choice != 0);
    return 0;
}
