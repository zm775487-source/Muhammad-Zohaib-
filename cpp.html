How the C++ Program Works

Setup()

Places the snake in the centre and drops food at a random position

Input()

Reads W A S D keys without stopping the game (kbhit / getch)

Logic()

Moves the tail, grows on food, checks wall & self collisions

Draw()

Draws into a screen buffer and updates only the changed rows on the console — smooth, flicker-free display (head O, body o, food F, walls #)

#include <iostream>
#include <conio.h>
#include <windows.h>
#include <cstdlib>
#include <ctime>
#include <cstdio>
#include <cstring>

using namespace std;

const int WIDTH = 40;
const int HEIGHT = 20;
const int MAX_TAIL = 100;

const int SCREEN_WIDTH = 60;
const int SCREEN_HEIGHT = 24;

int x, y, foodX, foodY, score;
int tailX[MAX_TAIL], tailY[MAX_TAIL];
int tailLen;

enum Direction
{
    STOP = 0,
    LEFT,
    RIGHT,
    UP,
    DOWN
};

Direction dir;
bool gameOver;

HANDLE consoleHandle;

char screen[SCREEN_HEIGHT][SCREEN_WIDTH];
char oldScreen[SCREEN_HEIGHT][SCREEN_WIDTH];

// Function declarations
void Setup();
void Draw();
void Input();
void Logic();
void GenerateFood();
void HideCursor();

// Hide cursor
void HideCursor()
{
    CONSOLE_CURSOR_INFO info;

    info.dwSize = 1;
    info.bVisible = FALSE;

    SetConsoleCursorInfo(consoleHandle, &info);
}

// Generate food
void GenerateFood()
{
    bool occupied;

    do
    {
        occupied = false;

        foodX = rand() % WIDTH;
        foodY = rand() % HEIGHT;

        if (foodX == x && foodY == y)
        {
            occupied = true;
        }

        for (int i = 0; i < tailLen; i++)
        {
            if (tailX[i] == foodX && tailY[i] == foodY)
            {
                occupied = true;
                break;
            }
        }

    } while (occupied);
}

// Setup game
void Setup()
{
    gameOver = false;
    dir = STOP;

    x = WIDTH / 2;
    y = HEIGHT / 2;

    score = 0;
    tailLen = 0;

    memset(oldScreen, 0, sizeof(oldScreen));

    GenerateFood();
}

// Draw game
void Draw()
{
    // Clear drawing buffer
    memset(screen, ' ', sizeof(screen));

    // Top and bottom border
    for (int j = 0; j < WIDTH + 2; j++)
    {
        screen[0][j] = '#';
        screen[HEIGHT + 1][j] = '#';
    }

    // Left and right border
    for (int i = 1; i <= HEIGHT; i++)
    {
        screen[i][0] = '#';
        screen[i][WIDTH + 1] = '#';
    }

    // Food
    screen[foodY + 1][foodX + 1] = 'F';

    // Snake tail
    for (int i = 0; i < tailLen; i++)
    {
        screen[tailY[i] + 1][tailX[i] + 1] = 'o';
    }

    // Snake head
    screen[y + 1][x + 1] = 'O';

    // Score
    char scoreText[50];

    sprintf(scoreText, "Score: %d", score);

    for (int i = 0; scoreText[i] != '\0'; i++)
    {
        screen[22][i] = scoreText[i];
    }

    // Controls
    const char controls[] = "W A S D = Move | X = Exit";

    for (int i = 0; controls[i] != '\0'; i++)
    {
        screen[23][i] = controls[i];
    }

    // Update only changed rows
    for (int i = 0; i < SCREEN_HEIGHT; i++)
    {
        if (memcmp(screen[i], oldScreen[i], SCREEN_WIDTH) != 0)
        {
            COORD position;

            position.X = 0;
            position.Y = i;

            DWORD written;

            WriteConsoleOutputCharacterA(
                consoleHandle,
                screen[i],
                SCREEN_WIDTH,
                position,
                &written
            );

            memcpy(oldScreen[i], screen[i], SCREEN_WIDTH);
        }
    }
}

// Input
void Input()
{
    if (_kbhit())
    {
        char key = _getch();

        switch (key)
        {
            case 'a':
            case 'A':
                if (dir != RIGHT)
                    dir = LEFT;
                break;

            case 'd':
            case 'D':
                if (dir != LEFT)
                    dir = RIGHT;
                break;

            case 'w':
            case 'W':
                if (dir != DOWN)
                    dir = UP;
                break;

            case 's':
            case 'S':
                if (dir != UP)
                    dir = DOWN;
                break;

            case 'x':
            case 'X':
                gameOver = true;
                break;
        }
    }
}

// Game logic
void Logic()
{
    // Move tail
    if (tailLen > 0)
    {
        for (int i = tailLen - 1; i > 0; i--)
        {
            tailX[i] = tailX[i - 1];
            tailY[i] = tailY[i - 1];
        }

        tailX[0] = x;
        tailY[0] = y;
    }

    // Move head
    switch (dir)
    {
        case LEFT:
            x--;
            break;

        case RIGHT:
            x++;
            break;

        case UP:
            y--;
            break;

        case DOWN:
            y++;
            break;

        default:
            break;
    }

    // Wall collision
    if (x < 0 || x >= WIDTH || y < 0 || y >= HEIGHT)
    {
        gameOver = true;
        return;
    }

    // Self collision
    for (int i = 0; i < tailLen; i++)
    {
        if (tailX[i] == x && tailY[i] == y)
        {
            gameOver = true;
            return;
        }
    }

    // Food collision
    if (x == foodX && y == foodY)
    {
        if (tailLen < MAX_TAIL)
        {
            // Add new tail segment
            if (tailLen == 0)
            {
                tailX[tailLen] = x - (dir == RIGHT) + (dir == LEFT);
                tailY[tailLen] = y - (dir == DOWN) + (dir == UP);
            }
            else
            {
                tailX[tailLen] = tailX[tailLen - 1];
                tailY[tailLen] = tailY[tailLen - 1];
            }

            tailLen++;
            score += 10;

            GenerateFood();
        }
        else
        {
            gameOver = true;
        }
    }
}

// Main function
int main()
{
    srand((unsigned int)time(0));

    // Set console size
    system("mode con: cols=65 lines=30");

    consoleHandle = GetStdHandle(STD_OUTPUT_HANDLE);

    // Clear screen once
    system("cls");

    HideCursor();

    Setup();

    while (!gameOver)
    {
        Input();

        if (gameOver)
            break;

        Logic();

        Draw();

        Sleep(100);
    }

    // Final result
    Draw();

    COORD position;

    position.X = 0;
    position.Y = 25;

    SetConsoleCursorPosition(consoleHandle, position);

    cout << "\nGAME OVER!" << endl;
    cout << "Final Score: " << score << endl;

    cout << "\nPress any key to exit...";

    _getch();

    return 0;
}
