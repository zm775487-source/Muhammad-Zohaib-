How the C++ Program Works

#include <iostream>
#include <conio.h>
#include <windows.h>
#include <cstdlib>
#include <ctime>

using namespace std;

const int WIDTH = 40;
const int HEIGHT = 20;
const int MAXLEN = WIDTH * HEIGHT;   // FIX: tail array now fits the whole board (800)

int x, y;
int foodX, foodY;
int score;
int tailX[MAXLEN], tailY[MAXLEN];    // FIX: was [100] - too small for a 40x20 board
int tailLen;

enum Direction { STOP = 0, LEFT, RIGHT, UP, DOWN };

Direction dir;
bool gameOver;

// Function declarations
void Setup();
void Draw();
void Input();
void Logic();
void GenerateFood();

void Setup()
{
    gameOver = false;
    dir = STOP;

    x = WIDTH / 2;
    y = HEIGHT / 2;

    score = 0;
    tailLen = 0;

    GenerateFood();
}

void GenerateFood()
{
    bool occupied;

    do
    {
        occupied = false;

        foodX = rand() % WIDTH;
        foodY = rand() % HEIGHT;

        // Check snake head
        if (foodX == x && foodY == y)
            occupied = true;

        // Check snake tail
        for (int i = 0; i < tailLen; i++)
        {
            if (tailX[i] == foodX && tailY[i] == foodY)
            {
                occupied = true;
                break;
            }
        }

    } while (occupied);
}

void Draw()
{
    system("cls");

    // Top border
    for (int i = 0; i < WIDTH + 2; i++)
        cout << "#";

    cout << endl;

    // Game area
    for (int i = 0; i < HEIGHT; i++)
    {
        for (int j = 0; j < WIDTH; j++)
        {
            if (j == 0)
                cout << "#";

            if (i == y && j == x)
            {
                cout << "O";
            }
            else if (i == foodY && j == foodX)
            {
                cout << "F";
            }
            else
            {
                bool tailPart = false;

                for (int k = 0; k < tailLen; k++)
                {
                    if (tailX[k] == j && tailY[k] == i)
                    {
                        cout << "o";
                        tailPart = true;
                        break;
                    }
                }

                if (!tailPart)
                    cout << " ";
            }

            if (j == WIDTH - 1)
                cout << "#";
        }

        cout << endl;
    }

    // Bottom border
    for (int i = 0; i < WIDTH + 2; i++)
        cout << "#";

    cout << "\nScore: " << score << endl;
    cout << "Controls: W = Up, S = Down, A = Left, D = Right, X = Exit" << endl;
}

void Input()
{
    if (_kbhit())
    {
        char key = _getch();

        switch (key)
        {
            case 'a':
            case 'A':
                if (dir != RIGHT)
                    dir = LEFT;
                break;

            case 'd':
            case 'D':
                if (dir != LEFT)
                    dir = RIGHT;
                break;

            case 'w':
            case 'W':
                if (dir != DOWN)
                    dir = UP;
                break;

            case 's':
            case 'S':
                if (dir != UP)
                    dir = DOWN;
                break;

            case 'x':
            case 'X':
                gameOver = true;
                break;
        }
    }
}

void Logic()
{
    // FIX (growth bug): save the position of the LAST tail segment
    // BEFORE the tail moves. When the snake eats food, the new
    // segment is placed exactly at this saved position.
    // (In the old code tailLen++ happened without ever setting the
    // new segment's position, so it appeared at (0,0) / a wrong
    // place for one frame - and the very first segment was never
    // placed at all, because with tailLen = 0 the tail never moved.)
    int lastX, lastY;
    if (tailLen > 0)
    {
        lastX = tailX[tailLen - 1];
        lastY = tailY[tailLen - 1];
    }
    else
    {
        // No tail yet: the first segment will be placed
        // where the head is right now
        lastX = x;
        lastY = y;
    }

    // Move tail
    if (tailLen > 0)
    {
        int prevX = tailX[0];
        int prevY = tailY[0];

        tailX[0] = x;
        tailY[0] = y;

        for (int i = 1; i < tailLen; i++)
        {
            int tempX = tailX[i];
            int tempY = tailY[i];

            tailX[i] = prevX;
            tailY[i] = prevY;

            prevX = tempX;
            prevY = tempY;
        }
    }

    // Move head
    switch (dir)
    {
        case LEFT:
            x--;
            break;

        case RIGHT:
            x++;
            break;

        case UP:
            y--;
            break;

        case DOWN:
            y++;
            break;

        default:
            break;
    }

    // Wall collision
    if (x < 0 || x >= WIDTH || y < 0 || y >= HEIGHT)
    {
        gameOver = true;
        return;
    }

    // Self collision
    for (int i = 0; i < tailLen; i++)
    {
        if (tailX[i] == x && tailY[i] == y)
        {
            gameOver = true;
            return;
        }
    }

    // Food collision
    if (x == foodX && y == foodY)
    {
        if (tailLen < MAXLEN)
        {
            // FIX: actually PLACE the new segment at the saved
            // position of the old last segment, then grow
            tailX[tailLen] = lastX;
            tailY[tailLen] = lastY;
            tailLen++;
            score += 10;
            GenerateFood();
        }
    }
}

int main()
{
    srand((unsigned int)time(0));

    Setup();

    while (!gameOver)
    {
        Draw();
        Input();
        Logic();

        Sleep(100);
    }

    system("cls");

    cout << "============================" << endl;
    cout << "         GAME OVER          " << endl;
    cout << "============================" << endl;

    cout << "Final Score: " << score << endl;

    cout << "\nPress any key to exit...";

    _getch();

    return 0;
}
