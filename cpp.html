// ============================================================
//  Snake Game - Complete C++ Project
//  Concepts: arrays, functions, loops, game loop,
//            random numbers, collision detection
//  Controls: W A S D to move, X to exit
// ============================================================
#include <iostream>
#include <conio.h>      // kbhit(), getch()  (Windows / Dev-C++)
#include <windows.h>    // Sleep()
#include <cstdlib>      // rand()
using namespace std;

const int WIDTH = 20, HEIGHT = 20;
int x, y, foodX, foodY, score;
int tailX[100], tailY[100], tailLen;
enum Direction { STOP = 0, LEFT, RIGHT, UP, DOWN };
Direction dir;
bool gameOver;

void Setup() {
    gameOver = false;
    dir = STOP;
    x = WIDTH / 2;  y = HEIGHT / 2;          // snake in centre
    foodX = rand() % WIDTH;                  // random food
    foodY = rand() % HEIGHT;
    score = 0;  tailLen = 0;
}

void Draw() {
    system("cls");
    for (int i = 0; i < WIDTH + 2; i++) cout << "#";
    cout << "\n";
    for (int i = 0; i < HEIGHT; i++) {
        for (int j = 0; j < WIDTH; j++) {
            if (j == 0) cout << "#";
            if (i == y && j == x) cout << "O";            // head
            else if (i == foodY && j == foodX) cout << "F"; // food
            else {
                bool tailPart = false;
                for (int k = 0; k < tailLen; k++)
                    if (tailX[k] == j && tailY[k] == i) {
                        cout << "o"; tailPart = true;
                    }
                if (!tailPart) cout << " ";
            }
            if (j == WIDTH - 1) cout << "#";
        }
        cout << "\n";
    }
    for (int i = 0; i < WIDTH + 2; i++) cout << "#";
    cout << "\nScore: " << score << "\n";
}

void Input() {
    if (_kbhit()) {
        switch (_getch()) {
            case 'a': dir = LEFT;  break;
            case 'd': dir = RIGHT; break;
            case 'w': dir = UP;    break;
            case 's': dir = DOWN;  break;
            case 'x': gameOver = true; break;
        }
    }
}

void Logic() {
    // move the tail: each segment follows the one before it
    int prevX = tailX[0], prevY = tailY[0], tempX, tempY;
    tailX[0] = x;  tailY[0] = y;
    for (int i = 1; i < tailLen; i++) {
        tempX = tailX[i];  tempY = tailY[i];
        tailX[i] = prevX;  tailY[i] = prevY;
        prevX = tempX;     prevY = tempY;
    }
    // move the head
    switch (dir) {
        case LEFT:  x--; break;
        case RIGHT: x++; break;
        case UP:    y--; break;
        case DOWN:  y++; break;
        default: break;
    }
    // wall collision
    if (x < 0 || x >= WIDTH || y < 0 || y >= HEIGHT) gameOver = true;
    // self collision
    for (int i = 0; i < tailLen; i++)
        if (tailX[i] == x && tailY[i] == y) gameOver = true;
    // food eaten: grow + new food + score
    if (x == foodX && y == foodY) {
        score += 10;
        tailLen++;
        foodX = rand() % WIDTH;
        foodY = rand() % HEIGHT;
    }
}

int main() {
    Setup();
    while (!gameOver) {
        Draw();
        Input();
        Logic();
        Sleep(80);          // game speed
    }
    cout << "Game Over! Final Score: " << score << "\n";
    return 0;
}
