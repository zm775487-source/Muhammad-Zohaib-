Stack (LIFO)

// Stack using an array — C++
class Stack {
  int arr[100], top;
public:
  Stack() { top = -1; }
  void push(int x) { arr[++top] = x; }
  int pop()  { return arr[top--]; }
  int peek() { return arr[top]; }
  bool empty() { return top == -1; }
};

Queue (FIFO)

// Queue using an array — C++
class Queue {
  int arr[100], front, rear;
public:
  Queue() { front = rear = -1; }
  void enqueue(int x) { arr[++rear] = x; }
  int dequeue() { return arr[++front]; }
  bool empty() { return front == rear; }
};

Bubble Sort

// Bubble sort — C++
void bubbleSort(int a[], int n) {
  for (int i = 0; i < n - 1; i++)
    for (int j = 0; j < n - i - 1; j++)
      if (a[j] > a[j + 1])
        swap(a[j], a[j + 1]);
}

Binary Search

// Binary search — O(log n), sorted array
int binarySearch(int a[], int n, int key) {
  int lo = 0, hi = n - 1;
  while (lo <= hi) {
    int mid = (lo + hi) / 2;
    if (a[mid] == key) return mid;
    if (a[mid] < key) lo = mid + 1;
    else hi = mid - 1;
  }
  return -1; // not found
}

Selection & Insertion Sort

// Selection sort — C++
void selectionSort(int a[], int n) {
  for (int i = 0; i < n - 1; i++) {
    int mn = i;
    for (int j = i + 1; j < n; j++)
      if (a[j] < a[mn]) mn = j;
    swap(a[i], a[mn]);
  }
}
// Insertion sort — C++
void insertionSort(int a[], int n) {
  for (int i = 1; i < n; i++) {
    int key = a[i], j = i - 1;
    while (j >= 0 && a[j] > key) {
      a[j + 1] = a[j]; j--;
    }
    a[j + 1] = key;
  }
}
