Stack (LIFO)

// Stack using an array — C++
class Stack {
  int arr[100], top;
public:
  Stack() { top = -1; }
  void push(int x) { arr[++top] = x; }
  int pop()  { return arr[top--]; }
  int peek() { return arr[top]; }
  bool empty() { return top == -1; }
};

Queue (FIFO)

// Queue using an array — C++
class Queue {
  int arr[100], front, rear;
public:
  Queue() { front = rear = -1; }
  void enqueue(int x) { arr[++rear] = x; }
  int dequeue() { return arr[++front]; }
  bool empty() { return front == rear; }
};

Singly Linked List

// Node + insert at head — C++
struct Node {
  int data;
  Node* next;
};
void insertHead(Node*& head, int x) {
  Node* n = new Node{x, head};
  head = n;
}

Bubble Sort

// Bubble sort — C++
void bubbleSort(int a[], int n) {
  for (int i = 0; i < n - 1; i++)
    for (int j = 0; j < n - i - 1; j++)
      if (a[j] > a[j + 1])
        swap(a[j], a[j + 1]);
}
